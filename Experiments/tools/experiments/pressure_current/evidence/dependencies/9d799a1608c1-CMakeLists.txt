cmake_minimum_required(VERSION 3.18)
project(KeyraReasonAdapter LANGUAGES CXX)
include(CTest)
option(KEYRA_BUILD_EXPERIMENTAL_FULL_MODEL "Build SDK-free tests of the full physical API adapter (not realtime or SDK qualified)" OFF)
set(KEYRA_RACK_FULL_MODEL_PROFILE "OriginalControl" CACHE STRING "Canonical full model build profile (candidate is opt-in)")
set_property(CACHE KEYRA_RACK_FULL_MODEL_PROFILE PROPERTY STRINGS OriginalControl FrozenConfidenceGrand)
# option() preserves an existing cache choice, including explicit OFF.
set(KEYRA_RACK_PRESSURE_SIMD_DEFAULT OFF)
if(KEYRA_RACK_FULL_MODEL_PROFILE STREQUAL "OriginalControl")
  set(KEYRA_RACK_PRESSURE_SIMD_DEFAULT ON)
endif()
option(KEYRA_RACK_FULL_PRESSURE_SIMD "Enable compiler-checked pressure SIMD in the experimental full native core only" ${KEYRA_RACK_PRESSURE_SIMD_DEFAULT})
option(KEYRA_RACK_FULL_FREE_UNISON_SIMD "Enable compiler-checked exact free/unison row SIMD in the OriginalControl native core only" ON)
option(KEYRA_RACK_FULL_FAST_CONTACT "Use the bounded Newton contact solve in experimental full cores" ON)
option(KEYRA_RACK_FULL_MECHANISM_CACHE "Cache equivalent mechanism/body force projection in experimental full cores" ON)
option(KEYRA_RACK_SANITIZERS "Instrument SDK-free adapter tests with address/undefined sanitizers" OFF)
if(KEYRA_RACK_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(KEYRA_INSTRUMENT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../Examples/Keyra/Instrument" CACHE PATH "Canonical Keyra checkout shared with the pinned Examples/Keyra submodule")
if(NOT EXISTS "${KEYRA_INSTRUMENT_DIR}/src/Piano.cpp" OR NOT EXISTS "${KEYRA_INSTRUMENT_DIR}/include/keyra/Piano.h")
  message(FATAL_ERROR "Set KEYRA_INSTRUMENT_DIR to the canonical virtual-instrument-keyra checkout. No DSP is copied into this adapter.")
endif()
if(NOT TARGET keyra_core)
  add_library(keyra_core STATIC "${KEYRA_INSTRUMENT_DIR}/src/Piano.cpp")
  target_include_directories(keyra_core PUBLIC "${KEYRA_INSTRUMENT_DIR}/include" PRIVATE "${KEYRA_INSTRUMENT_DIR}/src")
  target_compile_features(keyra_core PUBLIC cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(keyra_core PRIVATE -fno-exceptions -Wall -Wextra -Wpedantic)
  endif()
endif()
if(KEYRA_BUILD_EXPERIMENTAL_FULL_MODEL)
  if(NOT EXISTS "${KEYRA_INSTRUMENT_DIR}/src/full_instrument/FullInstrument.cpp" OR NOT EXISTS "${KEYRA_INSTRUMENT_DIR}/include/keyra/FullInstrument.h")
    message(FATAL_ERROR "The experimental full adapter requires the canonical FullInstrument API checkout.")
  endif()
  include("${KEYRA_INSTRUMENT_DIR}/src/full_instrument/FullModelProfile.cmake")
  keyra_resolve_full_profile("${KEYRA_RACK_FULL_MODEL_PROFILE}" KEYRA_RACK_SELECTED_PROFILE)
  set(KEYRA_RACK_PRESSURE_SIMD_ACTIVE 0)
  set(KEYRA_RACK_FREE_UNISON_SIMD_ACTIVE 0)
  set(KEYRA_RACK_FAST_CONTACT_ACTIVE $<BOOL:${KEYRA_RACK_FULL_FAST_CONTACT}>)
  set(KEYRA_RACK_MECHANISM_CACHE_ACTIVE $<BOOL:${KEYRA_RACK_FULL_MECHANISM_CACHE}>)
  # This optimization is ordinary C++, with no vector/compiler feature gate.
  # Target options follow containing-project flags so the chosen value wins
  # over a conflicting global definition, including the explicit opt-out.
  function(keyra_rack_select_mechanism_cache target)
    if(MSVC)
      target_compile_options(${target} PRIVATE
        /UKEYRA_FULL_MECHANISM_CACHE /DKEYRA_FULL_MECHANISM_CACHE=${KEYRA_RACK_MECHANISM_CACHE_ACTIVE})
    else()
      target_compile_options(${target} PRIVATE
        -UKEYRA_FULL_MECHANISM_CACHE -DKEYRA_FULL_MECHANISM_CACHE=${KEYRA_RACK_MECHANISM_CACHE_ACTIVE})
    endif()
  endfunction()
  # Match canonical policy for this independently compiled SDK-free target.
  # Jukebox/SDK builds never infer native compiler capability from this check.
  function(keyra_rack_select_simd_macro target macro active)
    if(MSVC)
      target_compile_options(${target} PRIVATE
        /U${macro} /D${macro}=${active})
    else()
      target_compile_options(${target} PRIVATE
        -U${macro} -D${macro}=${active})
    endif()
  endfunction()
  if(NOT MSVC AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND
      (KEYRA_RACK_FULL_PRESSURE_SIMD OR
       (KEYRA_RACK_FULL_FREE_UNISON_SIMD AND KEYRA_RACK_SELECTED_PROFILE_ID EQUAL 0)))
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("-fopenmp-simd" KEYRA_RACK_HAS_OPENMP_SIMD)
    if(KEYRA_RACK_HAS_OPENMP_SIMD)
      if(KEYRA_RACK_FULL_PRESSURE_SIMD)
        set(KEYRA_RACK_PRESSURE_SIMD_ACTIVE 1)
      endif()
      if(KEYRA_RACK_FULL_FREE_UNISON_SIMD AND KEYRA_RACK_SELECTED_PROFILE_ID EQUAL 0)
        set(KEYRA_RACK_FREE_UNISON_SIMD_ACTIVE 1)
      endif()
    endif()
  endif()
  # Keep the experiment's flags independent of a containing project's full
  # target. It always consumes canonical source, never a maintained DSP copy.
  add_library(keyra_rack_full_native_core STATIC "${KEYRA_INSTRUMENT_DIR}/src/full_instrument/FullInstrument.cpp")
  target_include_directories(keyra_rack_full_native_core PUBLIC "${KEYRA_INSTRUMENT_DIR}/include" PRIVATE "${KEYRA_INSTRUMENT_DIR}/src/full_instrument")
  target_compile_features(keyra_rack_full_native_core PUBLIC cxx_std_17)
  keyra_rack_select_mechanism_cache(keyra_rack_full_native_core)
  keyra_rack_select_simd_macro(keyra_rack_full_native_core KEYRA_FULL_FREE_UNISON_SIMD ${KEYRA_RACK_FREE_UNISON_SIMD_ACTIVE})
  keyra_rack_select_simd_macro(keyra_rack_full_native_core KEYRA_FULL_PRESSURE_SIMD ${KEYRA_RACK_PRESSURE_SIMD_ACTIVE})
  keyra_apply_full_profile(keyra_rack_full_native_core ${KEYRA_RACK_SELECTED_PROFILE_ID})
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(keyra_rack_full_native_core PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti -fno-builtin -ffp-contract=off
      -UKEYRA_FULL_FAST_CONTACT -DKEYRA_FULL_FAST_CONTACT=${KEYRA_RACK_FAST_CONTACT_ACTIVE})
    if(KEYRA_RACK_PRESSURE_SIMD_ACTIVE OR KEYRA_RACK_FREE_UNISON_SIMD_ACTIVE)
      target_compile_options(keyra_rack_full_native_core PRIVATE -fopenmp-simd)
    endif()
  else()
    target_compile_definitions(keyra_rack_full_native_core PRIVATE KEYRA_FULL_FAST_CONTACT=${KEYRA_RACK_FAST_CONTACT_ACTIVE})
  endif()
  message(STATUS "Experimental full pressure SIMD active: ${KEYRA_RACK_PRESSURE_SIMD_ACTIVE}")
  message(STATUS "Experimental full exact free/unison row SIMD active: ${KEYRA_RACK_FREE_UNISON_SIMD_ACTIVE}")
  message(STATUS "Experimental full bounded Newton contact: ${KEYRA_RACK_FULL_FAST_CONTACT}")
  message(STATUS "Experimental full mechanism/body projection cache: ${KEYRA_RACK_FULL_MECHANISM_CACHE}")
  message(STATUS "Experimental full profile: ${KEYRA_RACK_FULL_MODEL_PROFILE} (${KEYRA_RACK_SELECTED_PROFILE_ANCHORS} anchors, ${KEYRA_RACK_SELECTED_PROFILE_STRING_MODES} string modes)")
endif()
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(keyra_rack_host tests/WrapperHost.cpp Keyra.cpp JukeboxExports.cpp)
  target_include_directories(keyra_rack_host PRIVATE . tests/mock)
  target_link_libraries(keyra_rack_host PRIVATE keyra_core)
  target_compile_features(keyra_rack_host PRIVATE cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(keyra_rack_host PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions)
  endif()
  add_test(NAME keyra_reason_adapter COMMAND keyra_rack_host)
  add_test(NAME keyra_reason_lua_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/lua_contract.py")
  set_tests_properties(keyra_reason_lua_contract PROPERTIES ENVIRONMENT "KEYRA_INSTRUMENT_DIR=${KEYRA_INSTRUMENT_DIR}")
  if(KEYRA_BUILD_EXPERIMENTAL_FULL_MODEL)
    add_executable(keyra_rack_full_host tests/FullWrapperHost.cpp Keyra.cpp JukeboxExports.cpp)
    target_include_directories(keyra_rack_full_host PRIVATE . tests/mock)
    target_link_libraries(keyra_rack_full_host PRIVATE keyra_rack_full_native_core)
    target_compile_definitions(keyra_rack_full_host PRIVATE KEYRA_EXPERIMENTAL_FULL_MODEL=1)
    target_compile_definitions(keyra_rack_full_host PRIVATE KEYRA_EXPECT_FULL_STRING_MODES=${KEYRA_RACK_SELECTED_PROFILE_STRING_MODES})
    target_compile_features(keyra_rack_full_host PRIVATE cxx_std_17)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(keyra_rack_full_host PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti -fno-builtin -ffp-contract=off)
    endif()
    add_test(NAME keyra_reason_full_adapter COMMAND keyra_rack_full_host)
    foreach(rate IN ITEMS 44100 88200 96000)
      add_test(NAME keyra_reason_full_adapter_${rate} COMMAND keyra_rack_full_host ${rate})
    endforeach()
    add_test(NAME keyra_reason_full_lua_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/full_lua_contract.py")
    set_tests_properties(keyra_reason_full_lua_contract PROPERTIES ENVIRONMENT "KEYRA_INSTRUMENT_DIR=${KEYRA_INSTRUMENT_DIR}")
    if(NOT MSVC AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      add_test(NAME keyra_reason_full_build_policy COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/full_build_policy.py"
        --instrument-dir "${KEYRA_INSTRUMENT_DIR}" --compiler "${CMAKE_CXX_COMPILER}")
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT KEYRA_RACK_SANITIZERS)
      add_library(keyra_rack_full_scalar_core STATIC "${KEYRA_INSTRUMENT_DIR}/src/full_instrument/FullInstrument.cpp")
      target_include_directories(keyra_rack_full_scalar_core PUBLIC "${KEYRA_INSTRUMENT_DIR}/include" PRIVATE "${KEYRA_INSTRUMENT_DIR}/src/full_instrument")
      target_compile_features(keyra_rack_full_scalar_core PUBLIC cxx_std_17)
      keyra_rack_select_mechanism_cache(keyra_rack_full_scalar_core)
      # Scalar/unsupported/explicit-zero must retain the original schedule.
      keyra_rack_select_simd_macro(keyra_rack_full_scalar_core KEYRA_FULL_FREE_UNISON_SIMD 0)
      keyra_rack_select_simd_macro(keyra_rack_full_scalar_core KEYRA_FULL_PRESSURE_SIMD 0)
      keyra_apply_full_profile(keyra_rack_full_scalar_core ${KEYRA_RACK_SELECTED_PROFILE_ID})
      target_compile_options(keyra_rack_full_scalar_core PRIVATE -fno-exceptions -fno-rtti -fno-builtin -ffp-contract=off
        -UKEYRA_FULL_FAST_CONTACT -DKEYRA_FULL_FAST_CONTACT=${KEYRA_RACK_FAST_CONTACT_ACTIVE})
      if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(keyra_rack_full_scalar_core PRIVATE -fno-tree-vectorize)
      else()
        target_compile_options(keyra_rack_full_scalar_core PRIVATE -fno-vectorize -fno-slp-vectorize)
      endif()
      add_executable(keyra_rack_full_scalar_host tests/FullWrapperHost.cpp Keyra.cpp JukeboxExports.cpp)
      target_include_directories(keyra_rack_full_scalar_host PRIVATE . tests/mock)
      target_link_libraries(keyra_rack_full_scalar_host PRIVATE keyra_rack_full_scalar_core)
      target_compile_definitions(keyra_rack_full_scalar_host PRIVATE KEYRA_EXPERIMENTAL_FULL_MODEL=1)
      target_compile_definitions(keyra_rack_full_scalar_host PRIVATE KEYRA_EXPECT_FULL_STRING_MODES=${KEYRA_RACK_SELECTED_PROFILE_STRING_MODES})
      target_compile_features(keyra_rack_full_scalar_host PRIVATE cxx_std_17)
      target_compile_options(keyra_rack_full_scalar_host PRIVATE -fno-exceptions -fno-rtti -fno-builtin -ffp-contract=off)
      add_test(NAME keyra_reason_full_scalar_adapter COMMAND keyra_rack_full_scalar_host)
      foreach(rate IN ITEMS 44100 88200 96000)
        add_test(NAME keyra_reason_full_scalar_adapter_${rate} COMMAND keyra_rack_full_scalar_host ${rate})
      endforeach()
    endif()
  endif()
endif()
