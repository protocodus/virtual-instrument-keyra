# Optional official SDK adapter. The default core/CLAP build stays offline.
set(KEYRA_VST3_SDK_DIR "" CACHE PATH "Existing checkout of the official VST3 SDK (including its submodules)")
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
if(NOT KEYRA_VST3_SDK_DIR)
  include(FetchContent)
  FetchContent_Declare(keyra_vst3_sdk
    GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git
    GIT_TAG 3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96
    GIT_SUBMODULES base cmake pluginterfaces public.sdk
    GIT_SUBMODULES_RECURSE FALSE)
  FetchContent_MakeAvailable(keyra_vst3_sdk)
  set(KEYRA_VST3_SDK_DIR "${keyra_vst3_sdk_SOURCE_DIR}")
else()
  add_subdirectory("${KEYRA_VST3_SDK_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdk")
endif()
set(SDK_ROOT "${KEYRA_VST3_SDK_DIR}")
set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
set(pluginterfaces_SOURCE_DIR "${SDK_ROOT}/pluginterfaces")
smtg_add_vst3plugin(keyra_vst3 PACKAGE_NAME "Keyra" SOURCES_LIST KeyraVst3.cpp)
target_link_libraries(keyra_vst3 PRIVATE keyra_core sdk)
target_include_directories(keyra_vst3 PRIVATE "${SDK_ROOT}")
target_compile_features(keyra_vst3 PRIVATE cxx_std_17)
if(WIN32)
  target_compile_definitions(keyra_vst3 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(APPLE)
  smtg_target_set_bundle(keyra_vst3 BUNDLE_IDENTIFIER "com.protocodus.keyra" COMPANY_NAME "Keyra")
endif()
if(BUILD_TESTING AND TARGET validator)
  get_target_property(KEYRA_VST3_PACKAGE keyra_vst3 SMTG_PLUGIN_PACKAGE_PATH)
  add_test(NAME vst3_sdk_validator COMMAND validator "${KEYRA_VST3_PACKAGE}")
endif()
if(BUILD_TESTING AND (WIN32 OR (UNIX AND NOT APPLE)))
  add_executable(keyra_vst3_host host_test.cpp "${SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
  target_compile_features(keyra_vst3_host PRIVATE cxx_std_17)
  target_include_directories(keyra_vst3_host PRIVATE "${SDK_ROOT}")
  target_link_libraries(keyra_vst3_host PRIVATE keyra_core sdk_hosting ${CMAKE_DL_LIBS})
  if(WIN32)
    target_compile_definitions(keyra_vst3_host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  add_test(NAME vst3_host_state_and_events COMMAND keyra_vst3_host $<TARGET_FILE:keyra_vst3>)
endif()
