add_library(keyra_clap MODULE KeyraClap.cpp)
target_link_libraries(keyra_clap PRIVATE keyra_core)
target_include_directories(keyra_clap PRIVATE "${PROJECT_SOURCE_DIR}/third_party/clap/include")
target_compile_features(keyra_clap PRIVATE cxx_std_17)
set_target_properties(keyra_clap PROPERTIES PREFIX "" OUTPUT_NAME "Keyra" SUFFIX ".clap"
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(keyra_clap PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(APPLE)
  set_target_properties(keyra_clap PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION clap
    MACOSX_BUNDLE_GUI_IDENTIFIER io.protocodus.keyra)
endif()
install(TARGETS keyra_clap LIBRARY DESTINATION lib/clap RUNTIME DESTINATION lib/clap BUNDLE DESTINATION lib/clap)

if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(keyra_clap_host "${PROJECT_SOURCE_DIR}/tests/clap_host.cpp")
  target_include_directories(keyra_clap_host PRIVATE "${PROJECT_SOURCE_DIR}/third_party/clap/include")
  target_compile_features(keyra_clap_host PRIVATE cxx_std_17)
  target_link_libraries(keyra_clap_host PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
  if(KEYRA_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(keyra_clap_host PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(keyra_clap_host PRIVATE -fsanitize=address,undefined)
  endif()
  add_dependencies(keyra_clap_host keyra_clap)
  add_test(NAME clap_host_integration COMMAND keyra_clap_host $<TARGET_FILE:keyra_clap>)
endif()
