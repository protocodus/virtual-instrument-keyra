cmake_minimum_required(VERSION 3.18)
project(Keyra VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
option(KEYRA_BUILD_REFERENCE "Build the preserved offline research renderer" ON)

if(KEYRA_BUILD_REFERENCE)
  set(KEYRA_REFERENCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Research/GrandPianoV2")
  add_executable(keyra_reference "${KEYRA_REFERENCE_DIR}/src/grand.cpp")
  target_compile_features(keyra_reference PRIVATE cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(keyra_reference PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(BUILD_TESTING)
    add_test(NAME reference_self_test COMMAND keyra_reference --self-test)
    add_executable(keyra_reference_physics "${KEYRA_REFERENCE_DIR}/validation/v2_physics_probe.cpp")
    target_compile_features(keyra_reference_physics PRIVATE cxx_std_17)
    add_test(NAME reference_passivity_and_radiation COMMAND keyra_reference_physics)
  endif()
endif()
