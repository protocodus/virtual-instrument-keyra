cmake_minimum_required(VERSION 3.18)
project(Keyra VERSION 0.2.0 LANGUAGES CXX)

include(CTest)
option(KEYRA_BUILD_REFERENCE "Build the preserved offline research renderer" ON)
option(KEYRA_BUILD_FULL_REFERENCE "Build the sound-preserving optimized full physical renderer" ON)
option(KEYRA_BUILD_FULL_BACKEND "Build the block API for the full physical grand" ON)
option(KEYRA_FULL_PRESSURE_SIMD "Enable experimental supported four-lane pressure reductions in the native full piano API" OFF)
option(KEYRA_BUILD_PLUGIN "Build the CLAP instrument" ON)
option(KEYRA_BUILD_VST3 "Build the VST3 instrument (downloads pinned MIT SDK unless a path is provided)" OFF)
option(KEYRA_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(KEYRA_BUILD_PHYSICAL "Build isolated coupled physical-model research modules" ON)

add_library(keyra_core STATIC src/Piano.cpp)
target_compile_features(keyra_core PUBLIC cxx_std_17)
target_include_directories(keyra_core PUBLIC include PRIVATE src)
set_target_properties(keyra_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(keyra_core PRIVATE -Wall -Wextra -Wpedantic)
  if(KEYRA_SANITIZERS)
    target_compile_options(keyra_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(keyra_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()

if(KEYRA_BUILD_FULL_BACKEND)
  set(KEYRA_FULL_PRESSURE_SIMD_ENABLED OFF)
  add_library(keyra_full_instrument STATIC src/full_instrument/FullInstrument.cpp)
  target_compile_features(keyra_full_instrument PUBLIC cxx_std_17)
  target_include_directories(keyra_full_instrument PUBLIC include PRIVATE src/full_instrument)
  set_target_properties(keyra_full_instrument PROPERTIES POSITION_INDEPENDENT_CODE ON)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(keyra_full_instrument PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti -fno-builtin -ffp-contract=off)
    if(KEYRA_FULL_PRESSURE_SIMD AND NOT MSVC)
      include(CheckCXXCompilerFlag)
      check_cxx_compiler_flag("-fopenmp-simd" KEYRA_HAS_OPENMP_SIMD_FLAG)
      if(KEYRA_HAS_OPENMP_SIMD_FLAG)
        target_compile_options(keyra_full_instrument PRIVATE -fopenmp-simd)
        set(KEYRA_FULL_PRESSURE_SIMD_ENABLED ON)
        message(STATUS "Full piano pressure SIMD enabled (observer reductions only)")
      else()
        message(STATUS "Full piano pressure SIMD unavailable; retaining original pressure reductions")
      endif()
    endif()
    if(KEYRA_SANITIZERS)
      target_compile_options(keyra_full_instrument PUBLIC -O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(keyra_full_instrument PUBLIC -fsanitize=address,undefined)
    endif()
  elseif(MSVC)
    target_compile_options(keyra_full_instrument PRIVATE /fp:strict)
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND NOT MSVC)
    # Target options follow parent/global flags. Enforce the selected branch
    # even when a parent build defines this macro in CMAKE_CXX_FLAGS.
    target_compile_options(keyra_full_instrument PRIVATE -UKEYRA_FULL_PRESSURE_SIMD -DKEYRA_FULL_PRESSURE_SIMD=$<BOOL:${KEYRA_FULL_PRESSURE_SIMD_ENABLED}>)
  else()
    target_compile_definitions(keyra_full_instrument PRIVATE KEYRA_FULL_PRESSURE_SIMD=0)
  endif()
  add_executable(keyra_full_render tools/render_full.cpp)
  target_link_libraries(keyra_full_render PRIVATE keyra_full_instrument)
  target_compile_definitions(keyra_full_render PRIVATE KEYRA_RENDER_PRESSURE_SIMD=$<BOOL:${KEYRA_FULL_PRESSURE_SIMD_ENABLED}>)
  if(BUILD_TESTING)
    add_executable(keyra_full_instrument_tests tests/full_instrument_tests.cpp)
    target_link_libraries(keyra_full_instrument_tests PRIVATE keyra_full_instrument)
    add_test(NAME full_instrument COMMAND keyra_full_instrument_tests)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND NOT KEYRA_SANITIZERS)
      add_library(keyra_full_sdk_compat STATIC src/full_instrument/FullInstrument.cpp)
      target_compile_features(keyra_full_sdk_compat PUBLIC cxx_std_17)
      target_include_directories(keyra_full_sdk_compat PUBLIC include PRIVATE src/full_instrument)
      if(NOT MSVC)
        target_compile_options(keyra_full_sdk_compat PRIVATE -UKEYRA_FULL_PRESSURE_SIMD -DKEYRA_FULL_PRESSURE_SIMD=0)
      else()
        target_compile_definitions(keyra_full_sdk_compat PRIVATE KEYRA_FULL_PRESSURE_SIMD=0)
      endif()
      target_compile_options(keyra_full_sdk_compat PRIVATE -O3 -fno-exceptions -fno-rtti -fno-builtin -ffp-contract=off)
      if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(keyra_full_sdk_compat PRIVATE -fno-tree-vectorize)
      else()
        target_compile_options(keyra_full_sdk_compat PRIVATE -fno-vectorize -fno-slp-vectorize)
      endif()
      add_executable(keyra_full_sdk_compat_tests tests/full_instrument_tests.cpp)
      target_link_libraries(keyra_full_sdk_compat_tests PRIVATE keyra_full_sdk_compat)
      add_test(NAME full_sdk_portable COMMAND keyra_full_sdk_compat_tests)
    endif()
  endif()
endif()
